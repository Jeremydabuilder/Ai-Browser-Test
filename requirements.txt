# Phase 1 dependencies.
# PySide6 pulls in Qt WebEngine (Chromium), which does the actual page rendering.
# Everything else we need (sqlite3, pathlib, argparse) ships with CPython 3.11+.
PySide6>=6.6,<7

# Phase 2 (the AI agent). The browser itself runs without these; the agent
# panel reports that it is unconfigured if they are missing.
anthropic>=1.0,<2
keyring>=24            # OS keychain for the API key - never a file in the repo

# Groq and OpenRouter (free-tier-friendly alternatives to Anthropic - see
# Tools -> Configure AI Agent). httpx2 already ships as anthropic's own HTTP
# dependency; declared explicitly here because app/agent/openai_compatible.py
# imports it directly rather than relying on it being pulled in transitively.
httpx2

# PDF text extraction (app/browser/pdf_context.py), so Py can read a PDF's
# real text and page numbers instead of the rendered viewer plugin, which
# carries none of that as DOM. Pure Python, no OCR - a scanned PDF is
# reported as having no extractable text rather than silently OCR'd.
pypdf>=5,<7
# pypdf imports cryptography (for encrypted-PDF support) at import time,
# and on at least one environment this was tested in, the system
# `cryptography` package raised an unrecoverable Rust panic
# (ModuleNotFoundError: _cffi_backend) without it - not merely an
# ImportError Python code could catch. Declared explicitly so `pip
# install -r requirements.txt` alone is enough; nothing in this codebase
# imports cffi directly.
cffi
